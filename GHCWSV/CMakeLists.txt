cmake_minimum_required(VERSION 3.24)
project(GHCWSV LANGUAGES CXX CUDA)
find_package(Boost REQUIRED)
find_package(Threads REQUIRED)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 86 89 100 120)
endif()
add_executable(GHCWSV src/GHCWSV.cu)
set_target_properties(GHCWSV PROPERTIES CXX_STANDARD 17 CUDA_STANDARD 17 CUDA_RUNTIME_LIBRARY Static)
target_include_directories(GHCWSV PRIVATE ${Boost_INCLUDE_DIRS})
target_link_libraries(GHCWSV PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
if(MSVC)
  target_compile_options(GHCWSV PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8,/Zc:preprocessor>)
else()
  target_link_options(GHCWSV PRIVATE -static-libstdc++ -static-libgcc)
endif()
